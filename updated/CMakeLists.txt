cmake_minimum_required(VERSION 3.20)
project(FreehandOffline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(PkgConfig REQUIRED)
set(FREEHAND_OPENCV_PREFIX "" CACHE PATH "Optional private OpenCV usr prefix")
if(FREEHAND_OPENCV_PREFIX)
  set(OPENCV_INCLUDE_DIRS "${FREEHAND_OPENCV_PREFIX}/include/opencv4")
  set(OPENCV_LIBRARY_DIRS "${FREEHAND_OPENCV_PREFIX}/lib/x86_64-linux-gnu" "${FREEHAND_OPENCV_PREFIX}/lib")
else()
  pkg_check_modules(OPENCV REQUIRED opencv4)
endif()
pkg_check_modules(PNG REQUIRED IMPORTED_TARGET libpng)
find_library(OPENCV_CORE opencv_core HINTS ${OPENCV_LIBRARY_DIRS} REQUIRED)
find_library(OPENCV_IMGPROC opencv_imgproc HINTS ${OPENCV_LIBRARY_DIRS} REQUIRED)
option(FREEHAND_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(FREEHAND_KEEP_BOUNDARY_BUG "Regression ablation: retain original uninitialized boundary" OFF)
option(FREEHAND_KEEP_ZBUFFER_BUG "Regression ablation: retain original Zbuffer defects" OFF)
set(PREPARE_ARGS "${CMAKE_CURRENT_BINARY_DIR}/original-core")
if(FREEHAND_KEEP_BOUNDARY_BUG)
  list(APPEND PREPARE_ARGS --keep-boundary-bug)
endif()
if(FREEHAND_KEEP_ZBUFFER_BUG)
  list(APPEND PREPARE_ARGS --keep-zbuffer-bug)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sources.py" ${PREPARE_ARGS}
  RESULT_VARIABLE PREPARE_RESULT)
if(NOT PREPARE_RESULT EQUAL 0)
  message(FATAL_ERROR "Could not prepare original source build copies")
endif()
file(GLOB CORE_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/original-core/*.cpp")
add_library(freehand_core STATIC ${CORE_SOURCES})
target_include_directories(freehand_core PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/original-core" ${OPENCV_INCLUDE_DIRS})
target_compile_options(freehand_core PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/compat.h")
target_link_libraries(freehand_core PUBLIC ${OPENCV_IMGPROC} ${OPENCV_CORE})
add_executable(freehand_offline tools/offline.cpp)
target_link_libraries(freehand_offline PRIVATE freehand_core PkgConfig::PNG)
target_include_directories(freehand_offline PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../source/PointCloudSample/dinosaur/matrices")
if(FREEHAND_SANITIZE)
  foreach(target freehand_core freehand_offline)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
foreach(check boundary zbuffer carving marching-cubes)
  add_test(NAME ${check} COMMAND freehand_offline --test ${check})
endforeach()
