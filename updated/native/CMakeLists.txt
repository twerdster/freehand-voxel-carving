cmake_minimum_required(VERSION 3.20)
project(FreehandNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(FREEHAND_IOS_OPENCV_FRAMEWORK "" CACHE PATH "Official opencv2.framework directory for an Apple target")
if(FREEHAND_IOS_OPENCV_FRAMEWORK)
 set(OpenCV_INCLUDE_DIRS "${FREEHAND_IOS_OPENCV_FRAMEWORK}/Headers")
 set(OpenCV_LIBS "${FREEHAND_IOS_OPENCV_FRAMEWORK}/opencv2" "-framework Accelerate" "-framework CoreMedia" "-framework AVFoundation" "-framework CoreGraphics" "-framework CoreVideo" "-framework UIKit" "-lz")
elseif(FREEHAND_OPENCV_PREFIX)
 set(OpenCV_INCLUDE_DIRS "${FREEHAND_OPENCV_PREFIX}/include/opencv4")
 find_library(PRIVATE_OPENCV_CORE opencv_core PATHS "${FREEHAND_OPENCV_PREFIX}/lib/x86_64-linux-gnu" NO_DEFAULT_PATH REQUIRED)
 set(OpenCV_LIBS "${PRIVATE_OPENCV_CORE}")
else()
 find_package(OpenCV REQUIRED COMPONENTS core)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../prepare_sources.py" "${CMAKE_CURRENT_BINARY_DIR}/original-core" RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
 message(FATAL_ERROR "Original-core preparation failed")
endif()
file(GLOB sources "${CMAKE_CURRENT_BINARY_DIR}/original-core/*.cpp")
add_library(freehand_native STATIC freehand.cpp ${sources})
set_target_properties(freehand_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(freehand_native PUBLIC "${CMAKE_CURRENT_LIST_DIR}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/original-core" ${OpenCV_INCLUDE_DIRS})
target_compile_options(freehand_native PRIVATE -include "${CMAKE_CURRENT_LIST_DIR}/../compat.h")
if(FREEHAND_IOS_OPENCV_FRAMEWORK)
 get_filename_component(FRAMEWORK_PARENT "${FREEHAND_IOS_OPENCV_FRAMEWORK}" DIRECTORY)
 target_compile_options(freehand_native PRIVATE "-F${FRAMEWORK_PARENT}")
endif()
target_link_libraries(freehand_native PUBLIC ${OpenCV_LIBS})
option(WFC_BUILD_REPLAY "Build host PNG C-ABI replay" OFF)
if(WFC_BUILD_REPLAY)
 find_package(PNG REQUIRED)
 add_executable(freehand_native_replay replay.cpp)
 target_compile_definitions(freehand_native_replay PRIVATE PROJECT_FREEHAND=1)
 target_link_libraries(freehand_native_replay PRIVATE freehand_native PNG::PNG)
endif()
